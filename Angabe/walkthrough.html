Schritt 4 - PixelParser.Parse

WoCore/Pixel/PixelParser.cs (reine Logik, keine DB)
WarumDer Parser liefert Run-Records statt Entities - so ist er ohne Datenbank testbar (Core.Tests). Aufbau: Text in Zeilen teilen → #-Zeilen (Kommentar / #tags:) behandeln → leere Zeilen überspringen → pro Zeichen: Ziffer = Prefix aufbauen (prefix * 10 + ziffer, damit 12 funktioniert), . = Spalte weiterzählen, Großbuchstabe = Run, sonst ignorieren. Nach jedem Kommando wird der Prefix auf 0 gesetzt; unbekannte Zeichen setzen ihn nicht zurück.
Cheatsheet4.6 Import Schritt 1: Parser (S. 10-11) - Prefix-Logik ((prefix ?? 0) * 10 + (c - '0'), Reset nach jedem Kommando). Zeilen + Kommentare: 3.2 Text in Werte zerlegen (S. 4-5). Unterschied: hier gibt es keine Farbe $, dafür Spalte/Zeile mitzählen.
PixelParser.Parse
public static ParseResult Parse(string text)
{
    var tags  = new List<string>();
    var runs  = new List<Run>();
    int row   = 0;
    int width = 0;

    foreach (var rawLine in text.Split('\n'))
    {
        var line = rawLine.Trim();
        if (line.Length == 0)
        {
            continue; // empty lines are ignored
        }

        if (line.StartsWith('#'))
        {
            if (line.StartsWith(TagsHeader, StringComparison.OrdinalIgnoreCase))
            {
                tags.AddRange(line[TagsHeader.Length..]
                    .Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries));
            }

            continue; // other '#' lines are comments
        }

        int column = 0, prefix = 0;
        foreach (char c in line)
        {
            if (char.IsAsciiDigit(c))
            {
                prefix = prefix * 10 + (c - '0'); // multi digit prefix
                continue;
            }

            int count = prefix == 0 ? 1 : prefix;
            if (c == '.')
            {
                column += count; // transparent: no run
            }
            else if (char.IsAsciiLetterUpper(c))
            {
                runs.Add(new Run(row, column, count, c));
                column += count;
            }
            else
            {
                continue; // unknown character: ignore
            }

            prefix = 0; // reset after each command
        }

        width = Math.Max(width, column);
        row++;
    }

    return new ParseResult(tags, runs, width, row);
}

WoPersistence/ImportService.cs (+ Hilfsmethode GetOrCreateTagAsync)
Warum

  • Artist und Tags: vorhandene suchen, sonst neu anlegen (gleiches Muster wie das vorgegebene GetOrCreateArtistAsync).
  • Alles hängt über Navigation-Properties am Picture → ein AddAsync, keine Ids setzen, kein AddRange.

ImportService.ImportPictureAsync
public async Task ImportPictureAsync(string name, string? description, string? artistCode, DateOnly createdOn, string fileName)
{
    var result = PixelParser.Parse(await File.ReadAllTextAsync(fileName));

    using var trans = _uow.BeginTransaction();

    if (await _uow.PictureRepository.CountAsync(p => p.Name == name) > 0)
    {
        throw new InvalidOperationException($"Picture '{name}' already exists.");
    }

    var palette = (await _uow.PaletteColorRepository.GetAsync()).ToDictionary(c => c.Code[0]);
    var strokes = new List<Stroke>();
    foreach (var run in result.Runs)
    {
        if (!palette.TryGetValue(run.ColorCode, out var color))
        {
            throw new InvalidOperationException($"Unknown color code '{run.ColorCode}'.");
        }

        strokes.Add(new Stroke { Row = run.Row, Column = run.Column, Length = run.Length, PaletteColor = color });
    }

    var tags = new List<Tag>();
    foreach (var tagName in result.Tags.Distinct(StringComparer.OrdinalIgnoreCase))
    {
        tags.Add(await GetOrCreateTagAsync(tagName));
    }

    var picture = new Picture
    {
        Name        = name,
        Description = description,
        CreatedOn   = createdOn,
        Width       = result.Width,
        Height      = result.Height,
        Artist      = string.IsNullOrWhiteSpace(artistCode) ? null : await GetOrCreateArtistAsync(artistCode),
        Strokes     = strokes,
        Tags        = tags,
    };

    await _uow.PictureRepository.AddAsync(picture); // ONE Add: strokes, tags and artist via navigation properties
    await trans.CommitTransactionAsync();
}
ImportService.GetOrCreateTagAsync
private async Task<Tag> GetOrCreateTagAsync(string name)
{
    var tag = (await _uow.TagRepository.GetAsync(t => t.Name == name)).FirstOrDefault();
    if (tag == null)
    {
        tag = new Tag { Name = name };
        await _uow.TagRepository.AddAsync(tag);
    }

    return tag;
}

PictureRepository.GetOverviewAsync
public async Task<IList<PictureOverview>> GetOverviewAsync(int? artistId, string? tag)
{
    var query = DbSet.AsNoTracking();
    if (artistId != null)
    {
        query = query.Where(p => p.ArtistId == artistId);
    }

    if (!string.IsNullOrWhiteSpace(tag))
    {
        query = query.Where(p => p.Tags!.Any(t => t.Name.Contains(tag)));
    }

    return await query
        .OrderByDescending(p => p.CreatedOn)
        .ThenBy(p => p.Name)
        .Select(p => new PictureOverview(
            p.Id,
            p.CreatedOn,
            p.Name,
            p.ArtistId,
            p.Artist!.Name,
            p.Width,
            p.Height,
            p.Strokes!.Sum(s => s.Length),
            string.Join(",", p.Tags!.OrderBy(t => t.Name).Select(t => t.Name))))
        .ToListAsync();
}

Schritt 11 - Custom Draw: OnRender

WoWpf/Controls/PicturePreviewControl.cs (Dependency-Properties sind vorgegeben)
Warum
  • Die Bildgröße ist bekannt (PictureWidth/PictureHeight) - eine Bounding-Box wie im Cheatsheet ist nicht nötig.
  • Ein Faktor für x und y (Math.Min), sonst wird das Bild verzerrt; der Rest wird halbiert → zentriert.
  • Jeder Stroke ist ein Rechteck: x = Spalte, y = Zeile, Breite = Länge, Höhe = 1 Pixel - alles mal scale.
  • Brush aus dem Hex-String mit ColorConverter, pro Farbe nur einmal erzeugen und Freeze().
  • Show grid: senkrechte und waagrechte Linien im Abstand scale.
  • Neu gezeichnet wird automatisch: die Dependency-Properties haben AffectsRender.
Cheatsheet8.1 Teil A: UserControl mit bindbarer Property + Skalierung (S. 30-32) - Skalierung/Zentrierung (S. 31), dc.DrawRectangle(brush, pen, rect) (S. 32). Teil B (8.2 (S. 32)) entfällt: aus jedem Stroke wird direkt ein Rechteck statt einer Linie.
PicturePreviewControl.OnRender
protected override void OnRender(DrawingContext dc)
{
    // transparent background => the whole area is hit-testable and has a defined size
    dc.DrawRectangle(Brushes.Transparent, null, new Rect(0, 0, ActualWidth, ActualHeight));

    if (Strokes == null || PictureWidth <= 0 || PictureHeight <= 0)
    {
        return;
    }

    // uniform scaling, centered
    double scale   = Math.Min(ActualWidth / PictureWidth, ActualHeight / PictureHeight);
    double offsetX = (ActualWidth  - PictureWidth  * scale) / 2;
    double offsetY = (ActualHeight - PictureHeight * scale) / 2;

    var brushes = new Dictionary<string, Brush>();
    foreach (var stroke in Strokes)
    {
        var hex = stroke.PaletteColor?.Hex ?? "#FF00FF" /* unknown color */;
        if (!brushes.TryGetValue(hex, out var brush))
        {
            brush = new SolidColorBrush((Color)ColorConverter.ConvertFromString(hex));
            brush.Freeze();
            brushes[hex] = brush;
        }

        var rect = new Rect(
            offsetX + stroke.Column * scale,
            offsetY + stroke.Row    * scale,
            stroke.Length * scale,
            scale);
        dc.DrawRectangle(brush, null, rect);
    }

    // optional pixel grid
    if (ShowGrid)
    {
        var gridPen = new Pen(Brushes.LightGray, 0.5);
        gridPen.Freeze();
        for (int x = 0; x <= PictureWidth; x++)
        {
            dc.DrawLine(gridPen, new Point(offsetX + x * scale, offsetY),
                new Point(offsetX + x * scale, offsetY + PictureHeight * scale));
        }

        for (int y = 0; y <= PictureHeight; y++)
        {
            dc.DrawLine(gridPen, new Point(offsetX, offsetY + y * scale),
                new Point(offsetX + PictureWidth * scale, offsetY + y * scale));
        }
    }

    // frame of the picture area
    dc.DrawRectangle(null, new Pen(Brushes.LightGray, 1),
        new Rect(offsetX, offsetY, PictureWidth * scale, PictureHeight * scale));
}

PicturesController.GetAsync
/// <summary>
/// Get the details of a picture incl. tags and color statistics.
/// </summary>
/// <param name="id">id of the picture</param>
/// <response code="200">picture found</response>
/// <response code="404">no picture with this id</response>
[HttpGet("{id:int}")]
[ProducesResponseType(StatusCodes.Status200OK)]
[ProducesResponseType(StatusCodes.Status404NotFound)]
public async Task<ActionResult<PictureDetailDto>> GetAsync(int id)
{
    var picture = await _uow.PictureRepository.GetNoTrackingByIdAsync(id,
        nameof(Picture.Artist),
        nameof(Picture.Tags),
        $"{nameof(Picture.Strokes)}.{nameof(Stroke.PaletteColor)}");

    if (picture == null)
    {
        return NotFound();
    }

    var colors = picture.Strokes!
        .GroupBy(s => s.PaletteColor!.Code) // NoTracking: group by key, not by instance
        .Select(g => new ColorUsageDto(g.Key, g.First().PaletteColor!.Name, g.First().PaletteColor!.Hex, g.Sum(s => s.Length)))
        .OrderByDescending(c => c.Pixels)
        .ThenBy(c => c.Code)
        .ToList();

    return Ok(new PictureDetailDto(
        picture.Id,
        picture.Name,
        picture.Description,
        picture.CreatedOn,
        picture.Artist?.Name,
        picture.Width,
        picture.Height,
        picture.Tags!.Select(t => t.Name).OrderBy(t => t).ToList(),
        colors));
}
